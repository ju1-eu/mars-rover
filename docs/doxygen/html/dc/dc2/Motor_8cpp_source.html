33void driveSingleMotor(uint8_t pinFwd, uint8_t pinRev, int speed) {
34 // --- SAFETY: Clamping (Grenzwertbegrenzung) ---
35 // Verhindert, dass Werte außerhalb des PWM-Bereichs (z.B. >255)
36 // undefiniertes Verhalten in der SoftPWM-Lib auslösen.
37 if (speed > Config::SpeedMax) {
38 speed = Config::SpeedMax;
39 }
40 if (speed < -Config::SpeedMax) {
41 speed = -Config::SpeedMax;
42 }
43
44 // --- H-Bridge Logik ---
45 if (speed > 0) {
46 // Vorwärts: Fwd gepulst, Rev auf Ground (Low)
47 SoftPWMSet(pinFwd, speed);
48 SoftPWMSet(pinRev, 0);
49 } else if (speed < 0) {
50 // Rückwärts: Fwd auf Ground, Rev gepulst
51 // WICHTIG: speed ist hier negativ (z.B. -200), SoftPWM benötigt aber
52 // einen positiven Duty-Cycle. Daher: -speed (wird zu +200).
53 SoftPWMSet(pinFwd, 0);
54 SoftPWMSet(pinRev, -speed);
55 } else {
56 // Stopp / Neutral: Beide Leitungen auf Low -> Motor rollt aus
57 // (Coasting) Für aktives Bremsen müssten beide Pins auf HIGH (oder PWM
58 // Max) gehen, abhängig vom Motortreiber. Hier: Coasting gewählt.
59 SoftPWMSet(pinFwd, 0);
60 SoftPWMSet(pinRev, 0);
61 }
62}

76void init() {
77 // Startet den Timer-Interrupt für die Software-PWM
78 SoftPWMBegin();
79
80 // --- LATENCY OPTIMIZATION ---
81 // Standardmäßig hat SoftPWM eine "Fade-Time" (sanftes Anlaufen).
82 // In der Robotik übernimmt der Regelkreis (PID) die Beschleunigung.
83 // Der Treiber muss SOFORT reagieren (0ms Delay), sonst schwingt der Regler.
84 SoftPWMSetFadeTime(Pin::MotorL_Forward, 0, 0);
85 SoftPWMSetFadeTime(Pin::MotorL_Reverse, 0, 0);
86 SoftPWMSetFadeTime(Pin::MotorR_Forward, 0, 0);
87 SoftPWMSetFadeTime(Pin::MotorR_Reverse, 0, 0);
88
89 // Sicherer Startzustand
90 stop();
91}
